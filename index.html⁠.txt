  <script>
    // App State Initialization
    let appData = {
      hourly: [],
      daily: [],
      fuel: [],
      payouts: [],
      fuelSettings: { consumption: 8.5, price: 55 }
    };

    let currentCalendarDate = new Date();
    let selectedCalendarDay = null;

    function initApp() {
      // Load Data
      const savedData = localStorage.getItem('work_tracker_data');
      if (savedData) {
        appData = JSON.parse(savedData);
      }

      // Default Date Set to Today
      const today = new Date().toISOString().split('T')[0];
      document.getElementById('hourly-date').value = today;
      document.getElementById('daily-date').value = today;
      document.getElementById('fuel-date').value = today;
      document.getElementById('payout-date').value = today;

      // Load Fuel Settings
      if (appData.fuelSettings) {
        document.getElementById('fuel-consumption').value = appData.fuelSettings.consumption || '';
        document.getElementById('fuel-price').value = appData.fuelSettings.price || '';
      }

      renderAll();
    }

    function saveData() {
      localStorage.setItem('work_tracker_data', JSON.stringify(appData));
      renderAll();
    }

    function formatMoney(amount) {
      return new Intl.NumberFormat('ru-RU', { style: 'currency', currency: 'RUB', maximumFractionDigits: 0 }).format(amount);
    }

    function formatDate(dateStr) {
      if (!dateStr) return '';
      const parts = dateStr.split('-');
      return `${parts[2]}.${parts[1]}.${parts[0]}`;
    }

    function switchTab(tabId, title) {
      document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));

      document.getElementById(`tab-${tabId}`).classList.add('active');
      document.getElementById('screen-title').innerText = title;
      
      event.currentTarget.classList.add('active');

      if (tabId === 'calendar') {
        renderCalendar();
      } else if (tabId === 'payouts') {
        renderPayoutsStats();
      }
    }

    // --- Tab 1: Hourly ---
    function addHourly(e) {
      e.preventDefault();
      const item = {
        id: Date.now(),
        date: document.getElementById('hourly-date').value,
        hours: parseFloat(document.getElementById('hourly-hours').value),
        rate: parseFloat(document.getElementById('hourly-rate').value),
        note: document.getElementById('hourly-note').value || 'Почасовая работа'
      };
      appData.hourly.unshift(item);
      saveData();
      document.getElementById('hourly-note').value = '';
    }

    // --- Tab 2: Daily ---
    function addDaily(e) {
      e.preventDefault();
      const item = {
        id: Date.now(),
        date: document.getElementById('daily-date').value,
        shifts: parseFloat(document.getElementById('daily-shifts').value),
        rate: parseFloat(document.getElementById('daily-rate').value),
        note: document.getElementById('daily-note').value || 'Смена'
      };
      appData.daily.unshift(item);
      saveData();
      document.getElementById('daily-note').value = '';
    }

    // --- Tab 3: Fuel ---
    function saveFuelSettings() {
      appData.fuelSettings = {
        consumption: parseFloat(document.getElementById('fuel-consumption').value) || 0,
        price: parseFloat(document.getElementById('fuel-price').value) || 0
      };
      localStorage.setItem('work_tracker_data', JSON.stringify(appData));
    }

    function addFuel(e) {
      e.preventDefault();
      const distance = parseFloat(document.getElementById('fuel-distance').value);
      const consumption = appData.fuelSettings.consumption || 0;
      const price = appData.fuelSettings.price || 0;

      if (!consumption || !price) {
        alert('Пожалуйста, задайте средний расход и цену топлива в настройках выше.');
        return;
      }

      const total = (distance / 100) * consumption * price;

      const item = {
        id: Date.now(),
        date: document.getElementById('fuel-date').value,
        distance: distance,
        total: Math.round(total),
        note: document.getElementById('fuel-note').value || 'Поездка на авто'
      };

      appData.fuel.unshift(item);
      saveData();
      document.getElementById('fuel-distance').value = '';
      document.getElementById('fuel-note').value = '';
    }

    // --- Tab 5: Payouts ---
    function addPayout(e) {
      e.preventDefault();
      const item = {
        id: Date.now(),
        date: document.getElementById('payout-date').value,
        amount: parseFloat(document.getElementById('payout-amount').value),
        note: document.getElementById('payout-note').value || 'Выплата'
      };
      appData.payouts.unshift(item);
      saveData();
      document.getElementById('payout-amount').value = '';
      document.getElementById('payout-note').value = '';
    }

    function deleteItem(category, id) {
      appData[category] = appData[category].filter(item => item.id !== id);
      saveData();
    }

    // Rendering Helper
    function renderList(containerId, data, category, getTitle, getSubtitle, getAmount) {
      const container = document.getElementById(containerId);
      if (data.length === 0) {
        container.innerHTML = '<div style="color:var(--subtext-color); font-size:14px; text-align:center; padding:12px;">Записей нет</div>';
        return;
      }

      container.innerHTML = data.map(item => `
        <div class="list-item">
          <div class="list-info">
            <span class="list-date">${formatDate(item.date)}</span>
            <span class="list-desc">${getTitle(item)}</span>
            <span class="list-subtext">${getSubtitle(item)}</span>
          </div>
          <div class="list-right">
            <span class="list-amount">${formatMoney(getAmount(item))}</span>
            <button class="btn-delete" onclick="deleteItem('${category}', ${item.id})">×</button>
          </div>
        </div>
      `).join('');
    }

    function renderAll() {
      renderList('list-hourly', appData.hourly, 'hourly',
        i => i.note,
        i => `${i.hours} ч × ${i.rate} ₽`,
        i => i.hours * i.rate
      );

      renderList('list-daily', appData.daily, 'daily',
        i => i.note,
        i => `${i.shifts} смен(ы) × ${i.rate} ₽`,
        i => i.shifts * i.rate
      );

      renderList('list-fuel', appData.fuel, 'fuel',
        i => i.note,
        i => `${i.distance} км`,
        i => i.total
      );

      renderList('list-payouts', appData.payouts, 'payouts',
        i => i.note,
        i => 'Выплата от работодателя',
        i => i.amount
      );

      renderCalendar();
      renderPayoutsStats();
    }

    // --- Tab 4: Calendar Logic ---
    function changeMonth(delta) {
      currentCalendarDate.setMonth(currentCalendarDate.getMonth() + delta);
      selectedCalendarDay = null;
      document.getElementById('calendar-day-details-card').style.display = 'none';
      renderCalendar();
    }

    function renderCalendar() {
      const year = currentCalendarDate.getFullYear();
      const month = currentCalendarDate.getMonth();

      const monthNames = ["Январь", "Февраль", "Март", "Апрель", "Май", "Июнь", "Июль", "Август", "Сентябрь", "Октябрь", "Ноябрь", "Декабрь"];
      document.getElementById('calendar-month-year').innerText = `${monthNames[month]} ${year}`;

      const grid = document.getElementById('calendar-days');
      // Clear previous day cells (keeping 7 day headers)
      while (grid.children.length > 7) {
        grid.removeChild(grid.lastChild);
      }

      const firstDay = new Date(year, month, 1);
      const lastDay = new Date(year, month + 1, 0);

      // Starting day of the week (Monday = 0)
      let startingDay = firstDay.getDay() - 1;
      if (startingDay === -1) startingDay = 6;

      const monthDays = lastDay.getDate();

      // Empty cells before first day
      for (let i = 0; i < startingDay; i++) {
        const empty = document.createElement('div');
        empty.className = 'day-cell empty';
        grid.appendChild(empty);
      }

      let monthTotalSum = 0;

      // Populate Month Days
      for (let d = 1; d <= monthDays; d++) {
        const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
        
        const hasHourly = appData.hourly.some(i => i.date === dateStr);
        const hasDaily = appData.daily.some(i => i.date === dateStr);
        const hasFuel = appData.fuel.some(i => i.date === dateStr);

        // Sum earnings for this month
        appData.hourly.filter(i => i.date === dateStr).forEach(i => monthTotalSum += i.hours * i.rate);
        appData.daily.filter(i => i.date === dateStr).forEach(i => monthTotalSum += i.shifts * i.rate);
        appData.fuel.filter(i => i.date === dateStr).forEach(i => monthTotalSum += i.total);

        const cell = document.createElement('div');
        cell.className = 'day-cell';
        if (selectedCalendarDay === dateStr) cell.classList.add('selected');
        cell.innerText = d;

        // Render Dots
        if (hasHourly || hasDaily || hasFuel) {
          const dots = document.createElement('div');
          dots.className = 'day-dots';
          if (hasHourly) dots.innerHTML += '<div class="dot dot-hourly"></div>';
          if (hasDaily) dots.innerHTML += '<div class="dot dot-daily"></div>';
          if (hasFuel) dots.innerHTML += '<div class="dot dot-fuel"></div>';
          cell.appendChild(dots);
        }

        cell.onclick = () => selectCalendarDay(dateStr);
        grid.appendChild(cell);
      }

      document.getElementById('calendar-month-total').innerText = formatMoney(monthTotalSum);
    }

    function selectCalendarDay(dateStr) {
      selectedCalendarDay = dateStr;
      renderCalendar();

      const detailsCard = document.getElementById('calendar-day-details-card');
      const detailsContainer = document.getElementById('calendar-day-details');
      document.getElementById('calendar-selected-date-title').innerText = `Детали за ${formatDate(dateStr)}`;

      const hourlyItems = appData.hourly.filter(i => i.date === dateStr);
      const dailyItems = appData.daily.filter(i => i.date === dateStr);
      const fuelItems = appData.fuel.filter(i => i.date === dateStr);

      if (hourlyItems.length === 0 && dailyItems.length === 0 && fuelItems.length === 0) {
        detailsCard.style.display = 'none';
        return;
      }

      detailsCard.style.display = 'block';
      let html = '';

      hourlyItems.forEach(i => {
        html += `<div class="list-item"><div class="list-info"><span class="list-desc">⏱ ${i.note}</span><span class="list-subtext">${i.hours} ч × ${i.rate} ₽</span></div><div class="list-amount">${formatMoney(i.hours * i.rate)}</div></div>`;
      });

      dailyItems.forEach(i => {
        html += `<div class="list-item"><div class="list-info"><span class="list-desc">📅 ${i.note}</span><span class="list-subtext">${i.shifts} смена(ы) × ${i.rate} ₽</span></div><div class="list-amount">${formatMoney(i.shifts * i.rate)}</div></div>`;
      });

      fuelItems.forEach(i => {
        html += `<div class="list-item"><div class="list-info"><span class="list-desc">🚗 ${i.note}</span><span class="list-subtext">${i.distance} км</span></div><div class="list-amount">${formatMoney(i.total)}</div></div>`;
      });

      detailsContainer.innerHTML = html;
    }

    // --- Tab 5: Calculations ---
    function renderPayoutsStats() {
      const totalHourly = appData.hourly.reduce((sum, i) => sum + (i.hours * i.rate), 0);
      const totalDaily = appData.daily.reduce((sum, i) => sum + (i.shifts * i.rate), 0);
      const totalFuel = appData.fuel.reduce((sum, i) => sum + i.total, 0);

      const totalEarned = totalHourly + totalDaily + totalFuel;
      const totalPaid = appData.payouts.reduce((sum, i) => sum + i.amount, 0);
      const debt = totalEarned - totalPaid;

      document.getElementById('stat-total-earned').innerText = formatMoney(totalEarned);
      document.getElementById('stat-total-paid').innerText = formatMoney(totalPaid);
      
      const debtElement = document.getElementById('stat-debt');
      const debtCard = document.getElementById('debt-card');

      debtElement.innerText = formatMoney(debt);

      if (debt > 0) {
        debtCard.className = 'summary-card debt-red';
      } else {
        debtCard.className = 'summary-card debt-green';
      }
    }

    // Start App
    initApp();
